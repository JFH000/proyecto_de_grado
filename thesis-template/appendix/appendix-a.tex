\chapter{Notation}
\label{apx:notation}
% Tabla de los símbolos del documento agrupados por tema, con la ecuación o sección en que se definen, y aclaración de las letras con dos usos.

This appendix gathers the symbols of the document, grouped by topic, with the equation or section where each
one is defined. Some letters have two uses, which the context distinguishes:
\begin{itemize}[nosep]
  \item $\gamma$ with a subscript is an \acs{SINR}; alone, it is the discount factor;
  \item $\lambda$ alone is the arrival rate of data; with a subscript, it is a weight of the reward;
  \item $A$ alone is the penalty of the reward; $A(s, a)$ is the advantage of an action;
  \item $Q_k^t$ is a queue; $Q(s, a)$ is the value of an action;
  \item $P_k^t$ is a transmit power; $P(i)$ is the probability of drawing transition $i$;
  \item $p$ is the probability that two vehicles are neighbors; $p_i$ is the priority of transition $i$;
  \item $W$ is the bandwidth of a channel; $\bm{W}$ is the mixing matrix of an aggregation round;
  \item $e$ is the exponent of the path-loss law in Equation~\ref{eq:sm-channel} and the index of an episode
    in Section~\ref{sec:exp-runs};
  \item $n$ is the total amount of data in \acs{FedAvg} (Equation~\ref{eq:bg-fedavg}) and the number of
    comparisons in the sign test (Equation~\ref{eq:exp-sign-test}).
\end{itemize}

\notationgroup{Scenario}{
  $M$, $\Mset$ & Number and set of \acsp{CUE}, which are also the channels (15) & Sec.~\ref{sec:sm-scenario} \\
  $m$ & A \acs{CUE}, or its channel & Sec.~\ref{sec:sm-scenario} \\
  $K$, $\Kset$ & Number and set of \acs{V2V} pairs & Sec.~\ref{sec:sm-scenario} \\
  $k$, $k'$ & \acs{V2V} pairs, which are also the agents & Sec.~\ref{sec:sm-scenario} \\
  $t$, $\tau$ & Index of a slot and duration of a slot (\SI{1}{\milli\second}) & Sec.~\ref{sec:sm-scenario} \\
  $T$ & Horizon of the optimization problem, in slots & Eq.~\ref{eq:sm-problem} \\
  $m_k^t$ & Channel that pair $k$ chooses in slot $t$ & Sec.~\ref{sec:sm-scenario} \\
  $\rho_k^t[m]$ & 1 if pair $k$ chose channel $m$ in slot $t$, and 0 otherwise & Sec.~\ref{sec:sm-scenario} \\
  $P_k^t$, $\mathcal{P}$ & Transmit power of pair $k$ and set of possible levels & Eq.~\ref{eq:sm-power-levels} \\
  $N_p$ & Number of nonzero power levels (3) & Eq.~\ref{eq:sm-power-levels} \\
  $\Pmax$ & Maximum power of a pair (\SI{23}{dBm}) & Eq.~\ref{eq:sm-power-levels} \\
  $P^{\mathrm{c}}$ & Power of the \acsp{CUE} (\SI{23}{dBm}) & Sec.~\ref{sec:sm-scenario} \\
  $\bm{x}_k$ & Position of the transmitter of pair $k$ & Eq.~\ref{eq:sm-neighbors} \\
  $\Nset_k$ & Neighbors of pair $k$: the pairs whose transmitters are at most \SI{300}{\meter} away
    & Eq.~\ref{eq:sm-neighbors} \\
}

\notationgroup{Channel}{
  $P$, $h$, $I$, $R$ & Power, gain, interference, and rate of a generic link & Eq.~\ref{eq:bg-sinr} \\
  $h$, $\alpha$, $g$ & Gain of a link, its large-scale part, and its small-scale part & Eq.~\ref{eq:sm-channel} \\
  $d$, $d_{\mathrm{min}}$ & Distance of a link and minimum distance (\SI{1}{\meter}) & Eq.~\ref{eq:sm-channel} \\
  $\varphi$, $e$ & Constants of the path-loss law (\SI{-68.5}{dB} and 1.61) & Eq.~\ref{eq:sm-channel} \\
  $h_k^t[m]$ & Gain of the own link of pair $k$ on channel $m$ & Sec.~\ref{sec:sm-channel} \\
  $h_{m,\mathrm{B}}^t$, $h_{k,\mathrm{B}}^t$ & Gains toward the \acs{BS} from \acs{CUE} $m$ and from the
    transmitter of pair $k$ & Sec.~\ref{sec:sm-channel} \\
  $h_{m,k}^t$, $h_{k',k}^t$ & Gains toward the receiver of pair $k$ from \acs{CUE} $m$ and from the
    transmitter of pair $k'$ & Sec.~\ref{sec:sm-channel} \\
  $\alpha_k$, $\alpha_{k',k}$ & Large-scale gains of the own link and of an interfering link
    & Eq.~\ref{eq:sm-reliability} \\
  $\tilde h$ & Gain encoded for the neural network, $(10 \log_{10} h + 100) / 10$ & Eq.~\ref{eq:sm-observation} \\
  $\bar h_k^t$ & Mean observed gain of the own link over the $M$ channels, in dB (threshold rule)
    & Eq.~\ref{eq:exp-threshold} \\
  $\sigma^2$ & Noise power in a channel (\SI{-114}{dBm}) & Eq.~\ref{eq:sm-sinr-cue} \\
  $W$ & Bandwidth of a channel (\SI{1}{\mega\hertz}) & Eq.~\ref{eq:sm-rates} \\
}

  $\Rmin$ & Minimum rate of each \acs{CUE} (\SI{1}{Mbps}) & Eq.~\ref{eq:sm-problem} \\
  $\lambda$ & Arrival rate of data at each pair (\SI{1}{Mbps}) & Eq.~\ref{eq:sm-queue} \\
  $Q_k^t$ & Queue of pair $k$ at the start of slot $t$, in bits & Eq.~\ref{eq:sm-queue} \\
  $\Qmax$ & Maximum queue (\num{100000} bits) & Sec.~\ref{sec:sm-queue} \\
  $D_k^t$, $\Dmax$ & Delay of pair $k$ and maximum delay (\SI{100}{\milli\second}) & Sec.~\ref{sec:sm-queue} \\
